
#include <iostream>  // Библиотека для ввода/вывода
#include <math.h>    // Библиотека для математических функций 
#include <locale.h>  // Библиотека для настройки русского языка

using namespace std;
//Если убрать эту строчку, то перед каждым использованием стандартной функции или объекта нужно будет писать std::

int main() {
    //команда устанавливает русский язык для всех категорий
    setlocale(LC_ALL, "Russian");

    //Объявление переменных
    double a, b;      //для ввода двух чисел
    double max_num;   //для хранения большего числа
    double result;    //для результата функции
    //double число с плавающей точкой (тупи коти может забыть даже базу, поэтому помечаю для себя все:))

    //Ввод данных с приглашением к вводу
    cout << "Введите первое число: "; //выводит текст-приглашение на экран. Оператор << направляет текст в поток вывода.
    cin >> a;                         //oператор >> читает число, введенное пользователем с клавиатуры, и сохраняет его в переменную
    cout << "Введите второе число: ";
    cin >> b;

    //Поиск большего из двух чисел
    if (a > b) {
        max_num = a;
    }
    else {
        max_num = b;
    }

    //Вычисление функции F(x) = 2^x
    //Функция pow(основание, показатель_степени) из библиотеки <math.h> возводит первое число в степень второго.
    result = pow(2, max_num); //Результат сохраняется в переменную result

    //Вывод результата
    cout << "Большее число: " << max_num << endl; //endl (end line) — это манипулятор, который переводит курсор на новую строку
    cout << "Значение функции F(x) = 2^x для этого числа: " << result << endl;

    return 0; //Оператор return 0 сообщает операционной системе, что программа без ошибок


    //не бейте пж, я реал для себя любимой расписала все для понимания, а то будет как с циклом на зачете, что я просто забыла, а вы меня стукнете..:))
}
