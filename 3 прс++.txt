#include <iostream> //Для ввода/вывода 
#include <string> //Для работы со строками (string)
#include <locale.h> //Для русского языка в консоли

using namespace std;

//1 задание из 4 пр
void task1() {
    string text = "Программирование на C++"; //Исходная строка
    char sym;     //Искомый символ
    int count = 0;  //Счётчик найденных

    cout << "Введите символ: "; //Приглашение
    cin >> sym;    //Ввод символа

    for (int i = 0; i < (int)text.length(); i++) { //Проходим по строке
        if (text[i] == sym) count++;  //Если символ совпал — +1
    }

    if (count > 0)         //Если нашли хотя бы один
        cout << "Символ встречается " << count << " раз(а).\n";
    else             //Если не нашли
        cout << "Строка не содержит символ.\n";
}

//2 задание из пр
void task2() {
    string text = "Программирование на C++";   //Исходная строка
    char oldSym, newSym;                       //Что искать и на что менять
    bool found = false;                        //Флаг: была ли замена

    cout << "Введите символ для поиска: "; //Приглашение 1
    cin >> oldSym;             //Ввод символа 1
    cout << "Введите символ для замены: "; //Приглашение 2
    cin >> newSym;       //Ввод символа 2

    for (int i = 0; i < (int)text.length(); i++) { //Проходим по строке
        if (text[i] == oldSym) {       //Если нашли искомый символ
            text[i] = newSym;     //Заменяем его
            found = true;     //Ставим флаг
        }
    }

    if (found) {       //Если замена была
        cout << "Символ заменен.\n";
        cout << "Новая строка: " << text << "\n";
    }
    else {     //Если не нашли
        cout << "Символ не найден.\n";
    }
}

//3 задание из пр
void task3() {
    const int SIZE = 10; // Размер массива
    int arr[SIZE] = { 45, 12, 78, 3, 99, 21, 56, 34, 87, 10 }; //Массив

    //Сортировка пузырьком (по возрастанию)
    for (int i = 0; i < SIZE - 1; i++)  //Внешний цикл — проходы
        for (int j = 0; j < SIZE - 1 - i; j++) //Внутренний — сравнение пар
            if (arr[j] > arr[j + 1])      //Если левый больше правого
                swap(arr[j], arr[j + 1]);  //Меняем местами

    //Вывод отсортированного массива
    cout << "Отсортированный массив: ";
    for (int i = 0; i < SIZE; i++) cout << arr[i] << " ";
    cout << "\n";

    //Поиск числа
    int target;   //Искомое число
    cout << "Введите число для поиска: ";
    cin >> target;

    int pos = -1;     //-1 = не найдено
    for (int i = 0; i < SIZE; i++)     //Проходим по массиву
        if (arr[i] == target) {    //Если нашли
            pos = i;      //Запоминаем индекс
            break;     //Выходим из цикла
        }

    if (pos != -1)  //Если найдено
        cout << "Число найдено на позиции: " << pos + 1 << "\n";
    else   //Если не найдено
        cout << "Число не найдено.\n";
}
