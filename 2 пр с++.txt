#include <iostream>  //Подключаем библиотеку для ввода и вывода данных
#include <math.h>    //Подключаем библиотеку математических функций
#include <locale.h>  //Подключаем библиотеку для настройки русского языка в консоли

using namespace std; //Если убрать эту строчку, то перед каждым использованием стандартной функции или объекта нужно будет писать std::


// Эта функция запрашивает число, вычисляет F(x), спрашивает о продолжении.
// Если пользователь хочет продолжить — вызывает саму себя и суммирует результаты.
// Возвращает сумму всех вычисленных значений.
double processUser() { // функция для возвращения дробного числа )))))
    double x; //Переменная для хранения введённого числа
    double currentResult; //Переменная для результата текущего вычисления
    double totalSum; //Переменная для накопления общей суммы
    int continueChoice; //Переменная для выбора пользователя (1 - да, 0 - нет)

    cout << "Введите число x: "; //Выводим приглашение к вводу
    cin >> x; //Считываем число с клавиатуры

    currentResult = pow(2, x); //Вычисляем 2 в степени x

    cout << "Значение функции F(x) = 2^x для x = " << x << " равно: " << currentResult << endl; // объект класса из библиотеки (которая первая)

    cout << "Продолжить вычисления? (1 - да, 0 - нет): ";
    cin >> continueChoice;

    if (continueChoice == 1) {
        //кароче если пользователь хочет продолжить он это и делает - вызываем эту же функцию снова.
        //она должна вернуть сумму ВСЕХ последующих вычислений.
        //прибавляем её к нашему текущему результату и дай бог за работает
        totalSum = currentResult + processUser(); //РЕКУРСИВНЫЙ ВЫЗОВ (бл капс не выключила) 
    }
    else {
        //Если не дай бог пользователь отказался — возвращаем только текущий результат.
        totalSum = currentResult;
    }

    return totalSum; //Возвращаем накопленную сумму в main()
}


int main() {  // точка входа в программу (возвращает целое число -  мейн главная функция  ))))
    setlocale(LC_ALL, "Russian"); //Устанавливаем русский язык

    double finalSum; //Переменная для итоговой суммы

    cout << "Программа вычисления функции F(x) = 2^x" << endl;

    //Вызываем рекурсивную функцию и получаем общую сумму всех вычислений (пж работай)
    finalSum = processUser(); //Запуск рекурсии ( вызывает саму себя внутри своего тела для нее обязательно должно быть условие выхода)

    //Выводим итоговую сумму
    cout << "ИТОГОВАЯ СУММА всех вычисленных значений: " << finalSum << endl;
    cout << "Программа завершена. До свидания!" << endl;

    return 0; //Возвращаем 0 — спс что заработала :))
